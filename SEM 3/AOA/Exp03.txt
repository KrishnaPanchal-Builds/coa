To implement Binary Search for ‘n’ number and perform analysis using DAC Technique.

#include <stdio.h>

int comparisons = 0;

int binarySearch(int a[], int low, int high, int key)
{
    int mid;
    if (low > high)
        return -1;

    mid = (low + high) / 2;
    comparisons++;

    if (a[mid] == key)
        return mid;
    else if (key < a[mid])
        return binarySearch(a, low, mid - 1, key);
    else
        return binarySearch(a, mid + 1, high, key);
}

int main()
{
    int a[100], n, i, key, pos;

    printf("Enter number of elements: ");
    scanf("%d", &n);

    printf("Enter %d elements in sorted order: ", n);
    for (i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter element to search: ");
    scanf("%d", &key);

    pos = binarySearch(a, 0, n - 1, key);

    if (pos == -1)
        printf("Element not found\n");
    else
        printf("Element found at position %d\n", pos + 1);

    printf("Number of comparisons = %d\n", comparisons);

    /* Analysis (Divide and Conquer): T(n) = T(n/2) + 1
       Best case: O(1), Average and Worst case: O(log n) */

    return 0;
}
